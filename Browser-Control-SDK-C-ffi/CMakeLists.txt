cmake_minimum_required(VERSION 3.24)
project(BrowserControlFFI VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BCS_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/native" CACHE PATH "Canonical Browser SDK source")
add_subdirectory("${BCS_NATIVE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/native")
add_library(BrowserControlBridge SHARED BrowserControlBridge.cpp)
target_link_libraries(BrowserControlBridge PRIVATE BrowserControlSDK::sdk)
target_compile_features(BrowserControlBridge PRIVATE cxx_std_17)
target_compile_definitions(BrowserControlBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
