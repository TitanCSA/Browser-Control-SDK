cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(BrowserControl VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build with Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /INCREMENTAL:NO "/PDBALTPATH:%_PDB%")

set(WEBVIEW2_VERSION "1.0.4314.35")
set(WEBVIEW2_SHA256 "90778deb1b77e36292828b4b5833c9255c27533f40e0ddd55891ae718944edc4")
set(WEBVIEW2_CACHE "${CMAKE_SOURCE_DIR}/obj/Dependencies")
set(WEBVIEW2_ARCHIVE "${WEBVIEW2_CACHE}/microsoft.web.webview2.${WEBVIEW2_VERSION}.nupkg")
set(WEBVIEW2_ROOT "${WEBVIEW2_CACHE}/webview2-${WEBVIEW2_VERSION}")
file(MAKE_DIRECTORY "${WEBVIEW2_CACHE}")
if(NOT EXISTS "${WEBVIEW2_ARCHIVE}")
    file(DOWNLOAD
        "https://api.nuget.org/v3-flatcontainer/microsoft.web.webview2/${WEBVIEW2_VERSION}/microsoft.web.webview2.${WEBVIEW2_VERSION}.nupkg"
        "${WEBVIEW2_ARCHIVE}" EXPECTED_HASH "SHA256=${WEBVIEW2_SHA256}" TLS_VERIFY ON STATUS _download)
    list(GET _download 0 _download_code)
    if(NOT _download_code EQUAL 0)
        message(FATAL_ERROR "WebView2 SDK download failed. Check internet access or provide the pinned archive in obj/Dependencies.")
    endif()
endif()
file(SHA256 "${WEBVIEW2_ARCHIVE}" _webview_hash)
if(NOT _webview_hash STREQUAL WEBVIEW2_SHA256)
    message(FATAL_ERROR "Cached WebView2 SDK checksum differs from the reviewed pin.")
endif()
if(NOT EXISTS "${WEBVIEW2_ROOT}/build/native/include/WebView2.h")
    file(ARCHIVE_EXTRACT INPUT "${WEBVIEW2_ARCHIVE}" DESTINATION "${WEBVIEW2_ROOT}")
endif()

add_library(BrowserCommon INTERFACE)
target_include_directories(BrowserCommon SYSTEM INTERFACE "${CMAKE_SOURCE_DIR}/ThirdParty")
target_include_directories(BrowserCommon INTERFACE App)
add_executable(BrowserControl WIN32 App/Main.cpp App/Host.cpp App/Ipc.cpp)
target_include_directories(BrowserControl SYSTEM PRIVATE "${WEBVIEW2_ROOT}/build/native/include")
target_link_libraries(BrowserControl PRIVATE BrowserCommon
    "${WEBVIEW2_ROOT}/build/native/x64/WebView2LoaderStatic.lib"
    advapi32 bcrypt user32 gdi32 comctl32 shell32 ole32 oleaut32 shlwapi version dwmapi comdlg32)
target_link_options(BrowserControl PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
target_sources(BrowserControl PRIVATE App/BrowserControl.manifest)
if(EXISTS "${CMAKE_SOURCE_DIR}/SDK/Cpp/BrowserControl.cpp")
    add_library(BrowserControlSDK SDK/Cpp/BrowserControl.cpp)
    add_library(BrowserControlClient ALIAS BrowserControlSDK)
    target_include_directories(BrowserControlSDK PUBLIC SDK/Cpp)
    target_link_libraries(BrowserControlSDK PUBLIC BrowserCommon advapi32 shell32 ole32)
    if(EXISTS "${CMAKE_SOURCE_DIR}/Examples/cpp_basic.cpp")
        add_executable(BrowserControlExample Examples/cpp_basic.cpp)
        target_link_libraries(BrowserControlExample PRIVATE BrowserControlSDK)
    endif()
endif()
enable_testing()
if(EXISTS "${CMAKE_SOURCE_DIR}/Tests/IpcValidation.cpp")
    add_executable(BrowserIpcTests Tests/IpcValidation.cpp App/Ipc.cpp)
    target_link_libraries(BrowserIpcTests PRIVATE BrowserCommon advapi32 bcrypt shell32 ole32 user32)
    add_test(NAME PrivatePipeValidation COMMAND BrowserIpcTests)
    set_tests_properties(PrivatePipeValidation PROPERTIES TIMEOUT 35)
endif()
